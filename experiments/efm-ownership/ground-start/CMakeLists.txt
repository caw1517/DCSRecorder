cmake_minimum_required(VERSION 3.20)
project(GroundStartExperiment LANGUAGES CXX)
set(DCS_ROOT "D:/DCS World" CACHE PATH "Installed DCS build")
set(GROUND_TAPE_DIR "" CACHE PATH "Evidence-bound converted tape directory")
if(NOT EXISTS "${GROUND_TAPE_DIR}/token.txt" OR NOT EXISTS "${GROUND_TAPE_DIR}/recorded-flight.txt")
  message(FATAL_ERROR "Prepare and validate the real ground take first; supply GROUND_TAPE_DIR")
endif()
file(READ "${GROUND_TAPE_DIR}/token.txt" GROUND_TOKEN)
set(EFM "${CMAKE_CURRENT_SOURCE_DIR}/..")
enable_testing()
add_library(HornetGroundProbe SHARED "${EFM}/probe.cpp" "${EFM}/object_probe.cpp")
target_include_directories(HornetGroundProbe PRIVATE "${DCS_ROOT}/API/ExternalFMTemplate/ED_FM_Template" "${DCS_ROOT}/API/include")
target_compile_definitions(HornetGroundProbe PRIVATE HORNET_PROTOTYPE HORNET_RECORDED_PROTOTYPE HORNET_STAGED_PROTOTYPE HORNET_ENGINE_STAGED_PROTOTYPE HORNET_RELEASE_PROTOTYPE HORNET_GROUND_PROTOTYPE DCSR_GROUND_TAPE_TOKEN=${GROUND_TOKEN}ull DCSR_BUILD_2930 ED_FM_TEMPLATE_EXPORTS NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_features(HornetGroundProbe PRIVATE cxx_std_17)
set_property(TARGET HornetGroundProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(NAME IN ITEMS ground_path_check airborne_refusal_check)
  add_executable(${NAME} check_path.cpp)
  target_include_directories(${NAME} PRIVATE "${EFM}")
  target_compile_features(${NAME} PRIVATE cxx_std_17)
  add_test(NAME ${NAME} COMMAND ${NAME} "${GROUND_TAPE_DIR}/recorded-flight.txt")
endforeach()
target_compile_definitions(ground_path_check PRIVATE HORNET_GROUND_PROTOTYPE DCSR_GROUND_TAPE_TOKEN=${GROUND_TOKEN}ull)
add_executable(recorded_path_check "${EFM}/recorded_path_check.cpp")
target_compile_features(recorded_path_check PRIVATE cxx_std_17)
add_test(NAME airborne_path_regression COMMAND recorded_path_check)
add_executable(release_clock_check "${EFM}/release-start/check.cpp")
target_compile_features(release_clock_check PRIVATE cxx_std_17)
add_test(NAME release_clock_regression COMMAND release_clock_check)
add_executable(probe_check "${EFM}/probe_check.cpp")
target_include_directories(probe_check PRIVATE "${DCS_ROOT}/API/include")
target_compile_features(probe_check PRIVATE cxx_std_17)
add_test(NAME ground_callback_contract COMMAND probe_check $<TARGET_FILE:HornetGroundProbe> --owned-staged)
add_test(NAME ground_bridge_refusal COMMAND "${DCS_ROOT}/bin/luae.exe" "${EFM}/release-start/check_bridge.lua" $<TARGET_FILE:HornetGroundProbe>)
