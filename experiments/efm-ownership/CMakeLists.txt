cmake_minimum_required(VERSION 3.20)
project(DCSOwnershipProbe LANGUAGES CXX)
set(DCS_ROOT "D:/DCS World" CACHE PATH "Installed DCS root")
if(NOT EXISTS "${DCS_ROOT}/API/ExternalFMTemplate/ED_FM_Template/ED_FM_Template.cpp")
  message(FATAL_ERROR "DCS bundled EFM template is required")
endif()
add_library(OwnershipProbe SHARED probe.cpp object_probe.cpp)
target_include_directories(OwnershipProbe PRIVATE "${DCS_ROOT}/API/ExternalFMTemplate/ED_FM_Template")
target_include_directories(OwnershipProbe PRIVATE "${DCS_ROOT}/API/include")
target_compile_definitions(OwnershipProbe PRIVATE ED_FM_TEMPLATE_EXPORTS NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_features(OwnershipProbe PRIVATE cxx_std_17)
set_property(TARGET OwnershipProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(probe_check probe_check.cpp)
target_include_directories(probe_check PRIVATE "${DCS_ROOT}/API/include")
target_compile_features(probe_check PRIVATE cxx_std_17)
enable_testing()
add_test(NAME callback_contract COMMAND probe_check $<TARGET_FILE:OwnershipProbe>)
add_executable(turn_path_check turn_path_check.cpp)
target_compile_features(turn_path_check PRIVATE cxx_std_17)
add_test(NAME climbing_turn_geometry COMMAND turn_path_check)
